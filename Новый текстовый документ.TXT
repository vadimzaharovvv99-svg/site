        const track = document.getElementById('scroll-track');
        const heroOverlay = document.getElementById('hero-overlay');
        const sideLeft = document.getElementById('slide-left');
        const sideRight = document.getElementById('slide-right');
        const heroMainText = document.getElementById('hero-main-text');

        window.addEventListener('scroll', () => {
            const rect = track.getBoundingClientRect();
            let p = -rect.top / (rect.height - window.innerHeight);
            p = Math.max(0, Math.min(1, p)); 

            if (p < 0.15) {
                let phase = p / 0.15; 
                heroMainText.style.opacity = 1 - (phase * 0.5); 
                heroMainText.style.transform = `translate(-50%, -50%) scale(${1 - (0.05 * phase)})`;
                sideLeft.style.transform = `translateX(-100%)`;
                sideRight.style.transform = `translateX(100%)`;
                heroOverlay.style.opacity = 0;
            } 
            else if (p >= 0.15 && p < 0.70) {
                let phase = (p - 0.15) / 0.55;
                let ease = phase < 0.5 ? 2 * phase * phase : 1 - Math.pow(-2 * phase + 2, 2) / 2;
                heroMainText.style.opacity = 0.5 - (0.5 * ease);
                sideLeft.style.transform = `translateX(${-100 * (1 - ease)}%)`;
                sideRight.style.transform = `translateX(${100 * (1 - ease)}%)`;
                heroOverlay.style.opacity = 0;
            } 
            else if (p >= 0.70 && p < 0.90) {
                let phase = (p - 0.70) / 0.20;
                heroMainText.style.opacity = 0;
                sideLeft.style.transform = `translateX(0%)`;
                sideRight.style.transform = `translateX(0%)`;
                heroOverlay.style.opacity = phase;
            }
            else {
                heroMainText.style.opacity = 0;
                sideLeft.style.transform = `translateX(0%)`;
                sideRight.style.transform = `translateX(0%)`;
                heroOverlay.style.opacity = 1;
            }
        });

                if (entry.isIntersecting) {
                    entry.target.classList.add('active');
                }
            });
        }, { threshold: 0.15 });
        document.querySelectorAll('.reveal').forEach((el) => { observer.observe(el); });

                const eyeX = rect.left + rect.width / 2;
                const eyeY = rect.top + rect.height / 2;
                const deltaX = mouseX - eyeX;
                const deltaY = mouseY - eyeY;
                const angle = Math.atan2(deltaY, deltaX);
                const distance = Math.min(3, Math.hypot(deltaX, deltaY) / 10); 
                const pupilX = Math.cos(angle) * distance;
                const pupilY = Math.sin(angle) * distance;

                class Snowflake {
                    constructor() {
                        this.x = Math.random() * cWidth;
                        this.y = Math.random() * cHeight;
                        this.radius = Math.random() * 2.5 + 0.5;
                        this.speed = Math.random() * 3 + 1; 
                        this.wind = Math.random() * 1.5 - 0.75;

                    update() {
                        this.y += this.speed;
                        this.x += this.wind;
                        if (this.y > cHeight) {
                            this.y = -this.radius;
                            this.x = Math.random() * cWidth;
                        }

                        } else if (this.x < -this.radius) {

                        }
                    }
                    draw() {
                        ctx.beginPath();
                        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                        ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity})`;
                        ctx.fill();
                    }

            class Balls {
                constructor(context, buffer) { this.context = context; this.buffer = buffer; }
                setup() {
                    this.gainNode = this.context.createGain();
                    this.source = this.context.createBufferSource();
                    this.source.buffer = this.buffer;
                    this.source.connect(this.gainNode);
                    this.gainNode.connect(this.context.destination);
                    this.gainNode.gain.setValueAtTime(1, this.context.currentTime);
                }
                play() { this.setup(); this.source.start(this.context.currentTime); }
                stop() {
                    var ct = this.context.currentTime + 1;
                    this.gainNode.gain.exponentialRampToValueAtTime(.1, ct);
                    this.source.stop(ct);
                }
            }

            class Buffer {
                constructor(context, urls) { this.context = context; this.urls = urls; this.buffer = []; }
                loadSound(url, index) {
                    let request = new XMLHttpRequest();
                    request.open('get', url, true);
                    request.responseType = 'arraybuffer';
                    let thisBuffer = this;
                    request.onload = function() {
                        thisBuffer.context.decodeAudioData(request.response, function(buffer) {
                            thisBuffer.buffer[index] = buffer;
                            if(index == thisBuffer.urls.length-1) thisBuffer.loaded();
                        });
                    };
                    request.send();
                };
                getBuffer() { this.urls.forEach((url, index) => { this.loadSound(url, index); }) }
                loaded() {}
                getSound(index) { return this.buffer[index]; }
            }
